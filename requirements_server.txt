# ── PC / Laptop server dependencies ───────────────────────────────────────────
# Install with:  pip install -r requirements_server.txt
#
# This file covers both the AI server (Code/Server/) and the operator UI
# viewer (Code/Client/ai_viewer.py).  Both run on the same PC/laptop.
# ─────────────────────────────────────────────────────────────────────────────

# ── Deep learning (V-JEPA 2 world model) ──────────────────────────────────────
# torch 2.8+ bundles CUDA as hard deps on arm64 Linux (pulled by pip from PyPI).
# Cap at <2.8.0 to keep the Docker image CPU-only on Mac M-series arm64 Docker.
# The GPU build path (TORCH_INDEX build-arg) bypasses this cap via a separate install.
torch>=2.1.0,<2.8.0
torchvision>=0.16.0,<0.23.0  # required by ultralytics; V-JEPA 2 uses torch only
transformers>=4.40.0         # HuggingFace AutoModel for V-JEPA 2 weights
accelerate>=0.27.0           # fast model loading for transformers

# ── Object detection ──────────────────────────────────────────────────────────
# Needed for: demo mode (no robot) AND live mode with DETECTOR_LOCATION=server
# (the PC runs YOLO on the Pi's streamed frames). With DETECTOR_LOCATION=pi
# (default) the Pi runs YOLO and the server only receives CMD_DETECTION.
ultralytics>=8.1.0

# ── Computer vision ───────────────────────────────────────────────────────────
opencv-python>=4.8.0         # headless variant also works: opencv-python-headless
Pillow>=10.0.0               # required by ultralytics and transformers
numpy>=1.24.0

# ── Operator UI viewer ────────────────────────────────────────────────────────
# Streamlit viewer (Code/Client/streamlit_viewer.py) – browser-based, runs in Docker
streamlit>=1.37.0
# PyQt5 desktop viewer (Code/Client/ai_viewer.py) – native desktop use only
# Not installed in Docker (no arm64 wheel); run natively on Mac/Linux if preferred.
PyQt5>=5.15.0

# ── Config / utilities ────────────────────────────────────────────────────────
PyYAML>=6.0
