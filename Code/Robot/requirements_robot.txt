# ── Raspberry Pi robot client dependencies (thin client — NO AI) ──────────────
# Install with:  pip3 install -r requirements_robot.txt
#
# The Pi runs NO AI: all detection (YOLOv8n) and prediction (V-JEPA 2, SSv2) run
# on the PC server. So there is deliberately no torch / ultralytics here — the Pi
# only streams the camera, drives the motors, and reads the ultrasonic sensor.
#
# System packages must be installed first (run once on the Pi):
#   sudo apt-get update
#   sudo apt-get install -y \
#     python3-picamera2 python3-libcamera python3-kms++ \
#     python3-gpiozero python3-prctl libatlas-base-dev \
#     libopenblas-dev                     # speeds up numpy on arm64
# ─────────────────────────────────────────────────────────────────────────────

# ── Pi camera ─────────────────────────────────────────────────────────────────
# Prefer the apt package (python3-picamera2) which bundles libcamera bindings.
# This pip entry is a fallback for environments where apt is unavailable.
